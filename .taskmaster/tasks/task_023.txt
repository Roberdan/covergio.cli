3. Implement the MarkItDown agent class:
```typescript
import * as markitdown from '@microsoft/markitdown';

class MarkItDownAgent implements AgentInstance {
  id: string;
  definition: AgentDefinition;
  state: AgentState;
  memory: AgentMemory;
  
  constructor(id: string, memory: AgentMemory) {
    this.id = id;
    this.definition = markItDownAgentDefinition;
    this.state = { status: 'ready' };
    this.memory = memory;
  }
  
  async execute(input: string): Promise<AgentResponse> {
    // Parse the input to determine which MarkItDown functionality to use
    const command = this.parseCommand(input);
    
    try {
      switch (command.action) {
        case 'parseMarkdown':
          const parsed = markitdown.parse(command.content);
          return this.formatResponse(parsed);
          
        case 'extractHeadings':
          const document = markitdown.parse(command.content);
          const headings = document.filter(node => 
            node.type === 'heading'
          );
          return this.formatResponse(headings);
          
        case 'generateTableOfContents':
          const doc = markitdown.parse(command.content);
          const toc = markitdown.generateTableOfContents(doc, command.maxDepth || 3);
          return this.formatResponse(toc);
          
        case 'convertToHTML':
          const html = markitdown.renderHTML(command.content);
          return this.formatResponse(html);
          
        default:
          return {
            type: 'text',
            content: 'I can help with Markdown processing. Please specify an action like parseMarkdown, extractHeadings, generateTableOfContents, or convertToHTML.',
            metadata: { agentId: this.id }
          };
      }
    } catch (error) {
      return {
        type: 'error',
        content: `Error processing markdown: ${error.message}`,
        metadata: { agentId: this.id, error: error.toString() }
      };
    }
  }
  
  private parseCommand(input: string): any {
    // Implementation to parse user input into structured commands
    // This would extract the action and parameters from natural language
  }
  
  private formatResponse(data: any): AgentResponse {
    return {
      type: 'markdown',
      content: JSON.stringify(data, null, 2),
      metadata: { 
        agentId: this.id,
        timestamp: new Date().toISOString()
      }
    };
  }
}
```
