class VectorMemoryStore implements MemoryStore {
  constructor(private db: VectorDatabase) {}
  
  async store(agentId: string, memory: MemoryItem): Promise<string> {
    const vector = await this.embedMemory(memory.content);
    return this.db.insert({
      id: uuidv4(),
      agentId,
      vector,
      metadata: memory.metadata,
      content: memory.content
    });
  }
  
  async retrieve(agentId: string, query: string, limit = 5): Promise<MemoryItem[]> {
    const queryVector = await this.embedMemory(query);
    const results = await this.db.search(queryVector, { filter: { agentId }, limit });
    return results.map(r => ({
      id: r.id,
      content: r.content,
      metadata: r.metadata
    }));
  }
}
