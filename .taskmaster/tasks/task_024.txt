3. Implement the ImageAltText agent class:
```typescript
class ImageAltTextAgent implements AgentInstance {
  id: string;
  definition: AgentDefinition;
  state: AgentState;
  memory: AgentMemory;
  
  constructor(id: string, memory: AgentMemory) {
    this.id = id;
    this.definition = imageAltTextAgentDefinition;
    this.state = { status: 'ready' };
    this.memory = memory;
  }
  
  async execute(input: string): Promise<AgentResponse> {
    try {
      // Parse the markdown document
      const markitdown = require('@microsoft/markitdown');
      const document = await markitdown.parse(input);
      
      // Find all images without alt text or with placeholder alt text
      const imagesToProcess = document.findAll('image').filter(img => 
        !img.altText || img.altText.trim() === '' || 
        img.altText === 'image' || img.altText === 'placeholder'
      );
      
      // Process each image using LLM integration
      for (const image of imagesToProcess) {
        const altText = await this.generateAltText(image.url);
        image.altText = altText;
      }
      
      // Return the updated markdown
      return {
        type: 'success',
        content: document.toString(),
        metadata: {
          imagesProcessed: imagesToProcess.length
        }
      };
    } catch (error) {
      return {
        type: 'error',
        content: `Error processing images: ${error.message}`,
        metadata: { error: error.toString() }
      };
    }
  }
  
  private async generateAltText(imageUrl: string): Promise<string> {
    // Use MarkItDown's LLM integration for image description
    const markitdown = require('@microsoft/markitdown');
    const llmConfig = {
      provider: process.env.LLM_PROVIDER,
      apiKey: process.env.LLM_API_KEY,
      model: process.env.LLM_MODEL
    };
    
    // Generate descriptive alt text using the configured LLM
    const description = await markitdown.generateImageDescription(imageUrl, llmConfig);
    
    // Store in agent memory for future reference
    await this.memory.store({
      type: 'image-description',
      content: description,
      metadata: {
        imageUrl,
        timestamp: new Date().toISOString()
      }
    });
    
    return description;
  }
}
```

4. Register the agent with the AgentFactory:
```typescript
agentFactory.registerAgentType('image-alt-text', (id: string, memory: AgentMemory) => {
  return new ImageAltTextAgent(id, memory);
});
```
